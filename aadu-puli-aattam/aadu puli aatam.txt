class SoundManager {
  private sounds: Map<string, HTMLAudioElement> = new Map();
  private music: HTMLAudioElement | null = null;
  private volume = {
    master: 0.7,
    music: 0.5,
    sfx: 0.8
  };

  loadSound(name: string, path: string) {
    const audio = new Audio(path);
    audio.volume = this.volume.sfx * this.volume.master;
    this.sounds.set(name, audio);
  }

  playSound(name: string) {
    const sound = this.sounds.get(name);
    if (sound) {
      sound.currentTime = 0;
      sound.play().catch(e => console.log('Audio play failed:', e));
    }
  }

  playMusic(track: string) {
    if (this.music) {
      this.music.pause();
    }
    this.music = new Audio(track);
    this.music.loop = true;
    this.music.volume = this.volume.music * this.volume.master;
    this.music.play().catch(e => console.log('Music play failed:', e));
  }

  setVolume(type: 'master' | 'music' | 'sfx', value: number) {
    this.volume[type] = value;
    if (type === 'music' && this.music) {
      this.music.volume = this.volume.music * this.volume.master;
    }
  }

  mute() {
    this.volume.master = 0;
  }

  unmute() {
    this.volume.master = 0.7;
  }
}

function calculatePoints(puzzle: Puzzle, moves: number, hints: number): number {
  let basePoints = 100;
  
  // Deduct for extra moves
  const extraMoves = Math.max(0, moves - puzzle.parMoves);
  basePoints -= extraMoves * 5;
  
  // Deduct for hints used
  basePoints -= hints * 10;
  
  // Bonus for under par
  if (moves < puzzle.parMoves) {
    basePoints += (puzzle.parMoves - moves) * 10;
  }
  
  // Difficulty multiplier
  const multiplier = {
    easy: 1.0,
    medium: 1.5,
    hard: 2.0,
    expert: 3.0
  }[puzzle.difficulty];
  
  return Math.max(0, Math.floor(basePoints * multiplier));
}

      return;

    if (playerRole !== room.gameState.currentPlayer) {
      socket.emit('error', 'Not your turn');
      return;
    }
    
    // Validate move
    if (isValidMove(move, room.gameState)) {
      // Apply move
      room.gameState = applyMove(room.gameState, move);

      io.to(roomId).emit('moveMade', {
        move,
        gameState: room.gameState
      });
      
      // Check victory
      const winner = checkVictory(room.gameState);
      if (winner) {
        io.to(roomId).emit('gameOver', { winner });
      }
    } else {
      socket.emit('error', 'Invalid move');

    for (const [roomId, room] of rooms.entries()) {
      if (room.players.tiger === socket.id || room.players.goat === socket.id) {

const matchQueue: Array<{
  socketId: string;
  rating: number;
  timestamp: number;
}> = [];

function attemptMatch() {
  if (matchQueue.length < 2) return;
  
  // Sort by rating
  matchQueue.sort((a, b) => a.rating - b.rating);
  
  // Match closest ratings
  for (let i = 0; i < matchQueue.length - 1; i++) {
    const player1 = matchQueue[i];
    const player2 = matchQueue[i + 1];
    
    const ratingDiff = Math.abs(player1.rating - player2.rating);
    const waitTime = Date.now() - player1.timestamp;
    
    // Match if ratings close OR waited long time
    if (ratingDiff < 200 || waitTime > 60000) {
      createMatch(player1, player2);
      matchQueue.splice(i, 2);
      return;
    }
  }
}

function calculateEloChange(
  winnerRating: number,
  loserRating: number
): { winner: number; loser: number } {
  const K = 32; // K-factor
  
  const expectedWinner = 1 / (1 + Math.pow(10, (loserRating - winnerRating) / 400));
  const expectedLoser = 1 / (1 + Math.pow(10, (winnerRating - loserRating) / 400));
  
  const winnerChange = Math.round(K * (1 - expectedWinner));
  const loserChange = Math.round(K * (0 - expectedLoser));

  };
